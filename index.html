      // 3. STATE APLIKASI
      let activeCategory = 'Semua';

      // 4. UTILITAS KEAMANAN & FORMATTER
      function sanitizeInput(value, maxLength) {
        if (typeof value !== 'string') return '';
        return value.replace(/[<>]/g, '').trim().slice(0, maxLength);
      }

      function formatRupiah(amount) {
        const numeric = Number(amount) || 0;
        return new Intl.NumberFormat('id-ID', {
          style: 'currency',
          currency: 'IDR',
          minimumFractionDigits: 0,
          maximumFractionDigits: 0
        }).format(numeric);
      }

      function createEl(tag, classes, text) {
        const node = document.createElement(tag);
        if (classes) node.className = classes;
        if (text !== undefined && text !== null) {
          node.textContent = String(text);
        }
        return node;
      }

      function showToast(message, type = 'success') {
        const container = document.getElementById('toast-container');
        if (!container) return;

        const toast = createEl(
          'div',
          'pointer-events-auto animate-toast flex items-center gap-3 px-4 py-3 rounded-2xl shadow-xl border text-xs sm:text-sm font-semibold ' +
            (type === 'warning'
              ? 'bg-amber-50 border-amber-300 text-amber-900'
              : 'bg-[#005A32] border-[#C7E9C0] text-white')
        );

        const dot = createEl(
          'span',
          'w-2.5 h-2.5 rounded-full flex-shrink-0 ' + (type === 'warning' ? 'bg-[#D97706]' : 'bg-[#C7E9C0]')
        );
        const label = createEl('span', 'leading-snug', message);

        toast.appendChild(dot);
        toast.appendChild(label);
        container.appendChild(toast);

        setTimeout(() => {
          toast.style.opacity = '0';
          toast.style.transition = 'opacity 0.25s ease';
          setTimeout(() => {
            if (toast.parentNode) toast.parentNode.removeChild(toast);
          }, 260);
        }, 3000);
      }

      // 5. MANAJEMEN KERANJANG (LOCALSTORAGE DENGAN VALIDASI SKEMA)
      function loadCartFromStorage() {
        try {
          const raw = localStorage.getItem(STORAGE_KEY);
          if (!raw) return [];
          const parsed = JSON.parse(raw);
          if (!Array.isArray(parsed)) return [];
          // Validasi tiap item agar hanya ID valid dari katalog yang dimuat
          return parsed
            .filter((item) => item && typeof item.id === 'string' && Number.isInteger(item.qty) && item.qty > 0)
            .map((item) => {
              const found = CATALOG_PRODUCTS.find((p) => p.id === item.id);
              return found ? { id: found.id, qty: Math.min(item.qty, 100000) } : null;
            })
            .filter(Boolean);
        } catch (err) {
          return [];
        }
      }

      function saveCartToStorage() {
        try {
          localStorage.setItem(STORAGE_KEY, JSON.stringify(cart));
        } catch (err) {
          showToast('Penyimpanan lokal browser penuh.', 'warning');
        }
      }

      function addToCart(productId) {
        const product = CATALOG_PRODUCTS.find((p) => p.id === productId);
        if (!product) return;

        const existing = cart.find((item) => item.id === productId);
        if (existing) {
          existing.qty += product.minOrder;
        } else {
          cart.push({ id: product.id, qty: product.minOrder });
        }

        saveCartToStorage();
        updateCartUI();
        showToast(product.name + ' (+' + product.minOrder + ' ' + product.unit + ') ditambahkan.');
      }

      function updateCartQuantity(productId, delta) {
        const product = CATALOG_PRODUCTS.find((p) => p.id === productId);
        if (!product) return;

        const idx = cart.findIndex((item) => item.id === productId);
        if (idx === -1) return;

        const newQty = cart[idx].qty + delta;
        if (newQty <= 0) {
          cart.splice(idx, 1);
          showToast(product.name + ' dihapus dari keranjang.', 'warning');
        } else {
          cart[idx].qty = newQty;
        }

        saveCartToStorage();
        updateCartUI();
      }

      function renderCatalog() {

          const matchCat = activeCategory === 'Semua' || item.category === activeCategory;
          const matchText =
            item.name.toLowerCase().includes(keyword) ||
            item.description.toLowerCase().includes(keyword) ||
            item.category.toLowerCase().includes(keyword);
          return matchCat && matchText;
        });

        if (filtered.length === 0) {
          emptyState.classList.remove('hidden');
          return;
        }
        emptyState.classList.add('hidden');

        filtered.forEach((product) => {
          const card = createEl(

          badgeRow.appendChild(statusBadge);

          const specDot = createEl('span', 'w-2 h-2 rounded-full bg-[#005A32]');
          const specText = createEl('span', '', product.rendemen + ' • Min. ' + product.minOrder + ' ' + product.unit);
          specBox.appendChild(specDot);
          specBox.appendChild(specText);

          topSection.appendChild(badgeRow);
          topSection.appendChild(title);
          topSection.appendChild(desc);
          topSection.appendChild(specBox);

          // Bagian Bawah Kartu (Harga & Tombol)
          const bottomSection = createEl('div', 'pt-5 mt-5 border-t border-slate-100 flex items-center justify-between gap-3');

          const priceWrap = createEl('div');
          const priceLabel = createEl('span', 'block text-[11px] font-medium text-slate-400 uppercase', 'Harga Acuan');

          const priceLine = createEl('div');
          priceLine.appendChild(priceVal);
          priceLine.appendChild(unitVal);
          priceWrap.appendChild(priceLabel);
          priceWrap.appendChild(priceLine);

          const addBtn = createEl(
            'button',
            'px-4 py-2.5 rounded-xl bg-[#005A32] hover:bg-[#004224] text-white text-xs sm:text-sm font-bold shadow-sm transition-all active:scale-95 cursor-pointer',

          bottomSection.appendChild(addBtn);

          card.appendChild(topSection);

      // 7. RENDER UI KERANJANG BELANJA
      function updateCartUI() {
        const badge = document.getElementById('cart-badge');
        const container = document.getElementById('cart-items-container');
        const totalEl = document.getElementById('cart-total-price');
        if (!badge || !container || !totalEl) return;

        badge.textContent = String(cart.length);

        while (container.firstChild) {
          container.removeChild(container.firstChild);
        }

        if (cart.length === 0) {
          const emptyBox = createEl('div', 'text-center py-12 px-4');
          const emptyTitle = createEl('p', 'text-sm font-bold text-slate-700', 'Keranjang Anda masih kosong');
          const emptySub = createEl('p', 'text-xs text-slate-500 mt-1', 'Pilih TBS, bibit unggul, atau pupuk dari katalog untuk mulai menghitung estimasi.');
          emptyBox.appendChild(emptyTitle);
          emptyBox.appendChild(emptySub);
          container.appendChild(emptyBox);
          totalEl.textContent = formatRupiah(0);
          return;
        }

        let grandTotal = 0;

        cart.forEach((item) => {
          const product = CATALOG_PRODUCTS.find((p) => p.id === item.id);
          if (!product) return;

          const subtotal = product.price * item.qty;
          grandTotal += subtotal;

          const row = createEl('div', 'p-4 rounded-2xl bg-white border border-slate-200 shadow-xs space-y-3');

          const headerRow = createEl('div', 'flex items-start justify-between gap-2');
          const nameEl = createEl('h4', 'text-sm font-extrabold text-slate-900 leading-snug', product.name);
          const subEl = createEl('span', 'text-xs font-extrabold text-[#005A32] whitespace-nowrap', formatRupiah(subtotal));
          headerRow.appendChild(nameEl);
          headerRow.appendChild(subEl);

          const controlRow = createEl('div', 'flex items-center justify-between pt-1');
          const rateInfo = createEl('span', 'text-xs text-slate-500', formatRupiah(product.price) + ' / ' + product.unit);

          const stepper = createEl('div', 'inline-flex items-center gap-1.5 bg-slate-100 p-1 rounded-xl');
          const btnMinus = createEl('button', 'w-7 h-7 rounded-lg bg-white text-slate-700 font-bold text-sm shadow-2xs hover:bg-slate-200', '-');
          btnMinus.type = 'button';
          btnMinus.setAttribute('aria-label', 'Kurangi jumlah ' + product.name);
          btnMinus.addEventListener('click', () => updateCartQuantity(product.id, -product.minOrder));

          const qtyLabel = createEl('span', 'px-2 text-xs font-extrabold text-slate-800', item.qty + ' ' + product.unit);

          const btnPlus = createEl('button', 'w-7 h-7 rounded-lg bg-[#005A32] text-white font-bold text-sm shadow-2xs hover:bg-[#004224]', '+');
          btnPlus.type = 'button';
          btnPlus.setAttribute('aria-label', 'Tambah jumlah ' + product.name);
          btnPlus.addEventListener('click', () => updateCartQuantity(product.id, product.minOrder));

          stepper.appendChild(btnMinus);
          stepper.appendChild(qtyLabel);
          stepper.appendChild(btnPlus);

          controlRow.appendChild(rateInfo);
          controlRow.appendChild(stepper);

          row.appendChild(headerRow);
          row.appendChild(controlRow);
          container.appendChild(row);
        });

        totalEl.textContent = formatRupiah(grandTotal);
      }

      // 8. CHECKOUT WHATSAPP TERSTRUKTUR
      function handleWhatsAppCheckout() {
        if (cart.length === 0) {
          showToast('Keranjang masih kosong. Pilih komoditas terlebih dahulu.', 'warning');
          return;
        }

        const nameInput = document.getElementById('customer-name');
        const locInput = document.getElementById('customer-location');

        const customerName = sanitizeInput(nameInput ? nameInput.value : '', 50);
        const customerLocation = sanitizeInput(locInput ? locInput.value : '', 80);

        if (customerName.length < 2) {
          showToast('Mohon isi Nama Mitra / Pemesan minimal 2 karakter.', 'warning');
          if (nameInput) nameInput.focus();
          return;
        }

        if (customerLocation.length < 3) {
          showToast('Mohon isi Lokasi Kebun / Alamat Pengiriman.', 'warning');
          if (locInput) locInput.focus();
          return;
        }

        let grandTotal = 0;
        const lines = [
          'Halo Admin Niaga *Tualang Sawit*,',
          'Saya ingin melakukan konfirmasi pemesanan / transaksi komoditas sawit:',
          '',
          '*Data Mitra:*',
          '• Nama: ' + customerName,
          '• Lokasi: ' + customerLocation,
          '',
          '*Rincian Komoditas:*'
        ];

        cart.forEach((item, index) => {
          const product = CATALOG_PRODUCTS.find((p) => p.id === item.id);
          if (!product) return;
          const subtotal = product.price * item.qty;
          grandTotal += subtotal;
          lines.push(
            (index + 1) +
              '. ' +
              product.name +
              ' (' +
              item.qty +
              ' ' +
              product.unit +
              ' x ' +
              formatRupiah(product.price) +
              ') = *' +
              formatRupiah(subtotal) +
              '*'
          );
        });

        lines.push('');
        lines.push('*Total Estimasi: ' + formatRupiah(grandTotal) + '*');
        lines.push('Mohon informasi jadwal timbang / ketersediaan stok. Terima kasih.');

        const encodedMessage = encodeURIComponent(lines.join('\n'));
        const waUrl = 'https://wa.me/' + WA_BUSINESS_NUMBER + '?text=' + encodedMessage;

        window.open(waUrl, '_blank', 'noopener,noreferrer');
        showToast('Mengarahkan ke WhatsApp Niaga Tualang Sawit...');
      }

            const tempArea = document.createElement('textarea');
            tempArea.value = textToCopy;
            tempArea.style.position = 'fixed';
            tempArea.style.opacity = '0';
            document.body.appendChild(tempArea);
            tempArea.select();
            try {
              document.execCommand('copy');
              showToast('Berkas ' + currentCodeTab + ' berhasil disalin ke clipboard!');
            } catch (err) {
              showToast('Gagal menyalin kode.', 'warning');
            }
            document.body.removeChild(tempArea);
          });
        }
      }

      document.addEventListener('DOMContentLoaded', initApp);
    })();
